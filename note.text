import sounddevice as sd
import numpy as np
from scipy.signal import hilbert

# --- KONFIGURACE PRO BUFFER ---
# Zvětšíme blocksize na 16384 (cca 340ms při 48kHz)
# To dá procesoru obrovský čas na výpočet, ale zvuk bude opožděn
BLOCKSIZE = 16384 

def callback(indata, outdata, frames, time, status):
    if status:
        print(status)

    # Vstupní kanály
    L = indata[:, 0].astype(np.float32) / 32768.0
    R = indata[:, 1].astype(np.float32) / 32768.0

    # Hilbertova transformace na celém bloku
    # Používáme analytický signál k získání 90° posunu
    # np.imag(hilbert(x)) vytvoří verzi x posunutou o přesně 90 stupňů
    L_hilbert = np.imag(hilbert(L))
    R_hilbert = np.imag(hilbert(R))

    # SQ MATICE (Standardní rovnice)
    # Koeficienty pro SL a SR
    out_6ch = np.zeros((frames, 6), dtype=np.float32)
    
    # Přední kanály zůstávají
    out_6ch[:, 0] = L # FL
    out_6ch[:, 1] = R # FR
    
    # Zadní kanály (SQ dekódování vyžaduje fázové posuny)
    # SL = -0.707 * L_posunuto_o_90 + 0.707 * R
    # SR =  0.707 * L - 0.707 * R_posunuto_o_90
    out_6ch[:, 4] = (-0.707 * L_hilbert) + (0.707 * R)
    out_6ch[:, 5] = (0.707 * L) - (0.707 * R_hilbert)

    # Výstup zpět do int16
    out_6ch = np.clip(out_6ch * 0.8, -1.0, 1.0) # Headroom 0.8
    outdata[:] = (out_6ch * 32767.0).astype(np.int16)

# Spuštění s velkým bufferem
# latency='high' řekne systému, že nám nevadí zpoždění
with sd.Stream(channels=(2, 6), blocksize=BLOCKSIZE, 
               samplerate=48000, callback=callback, latency='high'):
    print("Dekódování běží s vysokou latencí... Stiskni Enter pro konec.")
    input()
Proč je to v reálném čase stále riskantní (proč zítra raději offline):
I s tímto bufferem uslyšíš v pravidelných intervalech (každých 16384 vzorků) jemné "škubnutí". Je to proto, že Hilbertův transformátor v scipy není navržen jako streamovací filtr. On se dívá na ten blok jako na izolovaný ostrov.

def create_sq_master(input_file, output_file):
    # 1. Načtení stereo souboru (např. Šavlový tanec)
    data, samplerate = sf.read(input_file)
    L = data[:, 0]
    R = data[:, 1]

    print(f"Zpracovávám: {input_file} ({samplerate}Hz)")

    # 2. Hilbertova transformace (90° posun) - OFFLINE (Celá skladba najednou)
    # np.imag(hilbert(x)) vytvoří matematicky čistý 90° posun v celém spektru
    L_90 = np.imag(hilbert(L))
    R_90 = np.imag(hilbert(R))

    # 3. SQ Dekódování do 5.1 (pro Onkyo HDMI)
